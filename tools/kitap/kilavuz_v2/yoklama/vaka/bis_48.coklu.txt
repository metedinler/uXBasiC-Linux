=====
Skaler VARPTR F64 ve U64
DIM x AS F64 = 12.5
DIM p AS PTR
p = VARPTR(x)
PRINT HEX(PEEKQ(p))
DIM u AS U64 = &H0123456789ABCDEF
DIM q AS PTR
q = VARPTR(u)
PRINT HEX(PEEKQ(q))
=====
VARPTR dizi elemani ve SIZEOF dizi
DIM a(0 TO 9) AS I32
DIM p1 AS PTR
DIM p2 AS PTR
p1 = VARPTR(a)
p2 = VARPTR(a(3))
PRINT p2 - p1
DIM b(10 TO 20) AS I32
DIM c1 AS PTR
DIM c2 AS PTR
c1 = VARPTR(b)
c2 = VARPTR(b(11))
PRINT c2 - c1
PRINT SIZEOF(a)

PRINT a(0)
