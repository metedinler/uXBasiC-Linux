S1 eleman takasi sayisal alanli
TYPE K
    a AS I32
    b AS I32
END TYPE
DIM k(2) AS K
DIM t AS K
DIM i AS I32
FOR i = 0 TO 2
    k(i).a = i
    k(i).b = i * 10
NEXT
t = k(0)
k(0) = k(1)
k(1) = t
PRINT k(0).a; " "; k(0).b; " | "; k(1).a; " "; k(1).b
=====
S2 eleman takasi STRING alanli
TYPE K
    ad AS STRING
    u AS I32
END TYPE
DIM k(2) AS K
DIM t AS K
DIM i AS I32
FOR i = 0 TO 2
    k(i).ad = "k" + STR(i)
    k(i).u = i * 10
NEXT
t = k(0)
k(0) = k(1)
k(1) = t
PRINT k(0).ad; " "; k(0).u; " | "; k(1).ad; " "; k(1).u
