=== instr_say_dongu
FUNCTION Say(metin AS STRING, parca AS STRING) AS I32
    DIM adet AS I32 = 0
    DIM konum AS I32 = 1
    DIM guvenlik AS I32 = 0
    DO
        konum = INSTR(konum, metin, parca)
        guvenlik += 1
        IF guvenlik > 20 THEN
            PRINT "SONSUZ"
            EXIT DO
        END IF
        IF konum = 0 THEN EXIT DO
        adet += 1
        konum += LEN(parca)
    LOOP
    RETURN adet
END FUNCTION
PRINT Say("[INFO] a [WARN] b [INFO] c", "[INFO]")
