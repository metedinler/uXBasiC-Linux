V1b genel TYPE degiskeni, LOCK, gorev icinden (beklenen 12 / 3)
TYPE Olcum
    kimlik AS I32
    deger AS I32
END TYPE
DIM son AS Olcum
THREAD Yaz(k AS I32)
    LOCK son
        son.kimlik = k
        son.deger = k * 4
    END LOCK
END THREAD
MAIN
    DIM t AS TASK = SPAWN Yaz(3)
    t.JOIN()
    PRINT son.deger
    PRINT son.kimlik
END MAIN

V1d dinamik sinirli TASK dizisi, PARALEL parametresiyle (beklenen 6)
THREAD Is(i AS I32) AS I32
    RETURN i
END THREAD
PARALEL Toplu(adet AS I32) AS I32
    DIM ts(adet - 1) AS TASK OF I32
    DIM i AS I32
    DIM t AS I32 = 0
    FOR i = 0 TO adet - 1
        ts(i) = SPAWN Is(i + 1)
    NEXT
    FOR i = 0 TO adet - 1
        t = t + ts(i).JOIN()
    NEXT
    RETURN t
END PARALEL
MAIN
    DIM b AS TASK OF I32 = SPAWN Toplu(3)
    PRINT b.JOIN()
END MAIN

V1f OPERATOR blok yazimi (tasarim)
CLASS V2
PUBLIC
    DIM x AS I32
    OPERATOR +(o AS V2) AS V2
        DIM r AS V2 = NEW V2()
        r.x = x + o.x
        RETURN r
    END OPERATOR
END CLASS
DIM a AS V2 = NEW V2()
DIM b AS V2 = NEW V2()
a.x = 1
b.x = 2
DIM c AS V2 = a + b
PRINT c.x

        RETURN 5
    END METHOD

    END METHOD
END CLASS
