H1 CLASS nesnesi SPAWN argumani (adim 5)
CLASS Kutu
PUBLIC
    DIM x AS I32
END CLASS
THREAD W(o AS Kutu) AS I32
    RETURN o.x
END THREAD
DIM a AS Kutu = NEW Kutu()
a.x = 3
DIM t AS TASK OF I32 = SPAWN W(a)
PRINT t.JOIN()
