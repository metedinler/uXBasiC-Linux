H6 dizi, kutuphane ile baslatilan DIM'den sonra
INCLUDE "include/modules/uxcollections.uxmh"
MAIN
    DIM anahtar AS U64 = uxcollections.ListNew()
    uxcollections.ListPushStr(anahtar, "a")
    DIM n AS I64 = uxcollections.ListCount(anahtar)
    DIM kul(7) AS I32
    DIM j AS I64
    FOR j = 0 TO n - 1
        kul(j) = 0
    NEXT
    FOR j = 0 TO n - 1
        IF kul(j) = 0 THEN PRINT "x"
    NEXT
END MAIN
