W2 OPERATOR blok yazimi, atama ayri satirda (beklenen 3)
CLASS Vek
PUBLIC
    DIM x AS I32
    OPERATOR +(o AS Vek) AS Vek
        DIM r AS Vek = NEW Vek()
        r.x = x + o.x
        RETURN r
    END OPERATOR
END CLASS
DIM a AS Vek = NEW Vek()
DIM b AS Vek = NEW Vek()
DIM c AS Vek
a.x = 1
b.x = 2
c = a + b
PRINT c.x
