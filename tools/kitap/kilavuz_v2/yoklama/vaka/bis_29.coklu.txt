G1 LIST argumani SPAWN sinirinda (adim 5)
THREAD W(l AS LIST) AS I32
    RETURN LISTLEN(l)
END THREAD
DIM l AS LIST
LISTADD(l, 1)
DIM t AS TASK OF I32 = SPAWN W(l)
PRINT t.JOIN()
=====
G2 CLASS nesnesi SPAWN argumani (adim 5)
CLASS K
PUBLIC
    DIM x AS I32
END CLASS
THREAD W(k AS K) AS I32
    RETURN k.x
END THREAD
DIM a AS K = NEW K()
a.x = 3
DIM t AS TASK OF I32 = SPAWN W(a)
PRINT t.JOIN()
