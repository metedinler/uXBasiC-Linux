N1a sinif alani LIST, kurucuda LIST_NEW (beklenen 2 / armut)
CLASS Sepet
    PUBLIC
    DIM urunler AS LIST
    CONSTRUCTOR()
        urunler = LIST_NEW()
    END CONSTRUCTOR
    METHOD Ekle(u AS STRING)
        LISTADD(urunler, u)
    END METHOD
    METHOD Say() AS I32
        RETURN LISTLEN(urunler)
    END METHOD
END CLASS
DIM s AS Sepet = NEW Sepet()
s.Ekle("elma")
s.Ekle("armut")
PRINT s.Say()
DIM u AS STRING
FOR EACH u IN s.urunler
    PRINT u
NEXT

CLASS Defter
    PUBLIC
    DIM kayit AS DICT
    CONSTRUCTOR()
        kayit = DICT_NEW()
    END CONSTRUCTOR
    METHOD Yaz(k AS STRING, v AS I32)
        DICTSET(kayit, k, v)
    END METHOD
    METHOD Say() AS I32
        RETURN DICTLEN(kayit)
    END METHOD
END CLASS
DIM d AS Defter = NEW Defter()
d.Yaz("a", 5)
d.Yaz("b", 6)
PRINT d.Say()
PRINT DICTGET(d.kayit, "a")
