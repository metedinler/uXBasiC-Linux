    DIM s AS STRING
    DIM f AS F64

    END METHOD

    END METHOD
END CLASS

J4 THIS.alan = parametre, alan adi __ ile baslar
CLASS Ak
PRIVATE
    __SV AS STRING
    __FV AS F64
PUBLIC
    METHOD SetF(v AS F64)
        THIS.__FV = v
    END METHOD
    METHOD GetF() AS F64
        RETURN THIS.__FV
    END METHOD
END CLASS
DIM a AS Ak = NEW Ak()
a.SetF(2.5)
PRINT a.GetF()
