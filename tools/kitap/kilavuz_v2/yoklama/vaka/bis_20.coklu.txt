FUNCTION Uret(n AS I32) AS LIST
    DIM l AS LIST
    DIM i AS I32
    FOR i = 1 TO n
        LISTADD(l, i * i)
    NEXT
    RETURN l
END FUNCTION
DIM sonuc AS LIST
sonuc = Uret(4)
PRINT LISTLEN(sonuc)
PRINT LISTGET(sonuc, 3)
