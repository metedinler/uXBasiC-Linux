S198-3 MAGIC ADD (beklenen 4)
CLASS Sayi
    PUBLIC:
    DIM v AS I32
    MAGIC ADD(other AS Sayi) AS Sayi
        DIM r AS Sayi = NEW Sayi()
        r.v = v + other.v
        RETURN r
    END MAGIC
END CLASS

DIM a AS Sayi = NEW Sayi()
DIM b AS Sayi = NEW Sayi()
DIM c AS Sayi = NEW Sayi()
a.v = 1
b.v = 3
c = a + b
PRINT c.v
