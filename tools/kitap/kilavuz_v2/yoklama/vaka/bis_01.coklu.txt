T1 fonksiyon param TYPE ic erisim
TYPE Nokta
    x AS I32
    y AS I32
END TYPE
TYPE Cizgi
    a AS Nokta
    b AS Nokta
END TYPE
FUNCTION Uzunluk2(c AS Cizgi) AS I32
    DIM dx AS I32 = c.b.x - c.a.x
    DIM dy AS I32 = c.b.y - c.a.y
    RETURN dx * dx + dy * dy
END FUNCTION
DIM k AS Cizgi
k.a.x = 0: k.a.y = 0: k.b.x = 3: k.b.y = 4
PRINT Uzunluk2(k)
