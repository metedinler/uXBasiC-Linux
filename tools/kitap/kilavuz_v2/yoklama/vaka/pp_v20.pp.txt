=== v4_dongu_ayri_degisken
FUNCTION Say(metin AS STRING, parca AS STRING) AS I32
    DIM adet AS I32 = 0
    DIM bas AS I32 = 1
    DIM r AS I32
    DIM g AS I32 = 0
    DO
        r = INSTR(bas, metin, parca)
        g += 1
        IF g > 10 THEN
            PRINT "SONSUZ"
            EXIT DO
        END IF
        IF r = 0 THEN EXIT DO
        adet += 1
        bas = r + LEN(parca)
    LOOP
    RETURN adet
END FUNCTION
PRINT Say("[INFO] a [WARN] b [INFO] c", "[INFO]")
=== v5_do_while
FUNCTION Say(metin AS STRING, parca AS STRING) AS I32
    DIM adet AS I32 = 0
    DIM konum AS I32 = 1
    konum = INSTR(konum, metin, parca)
    DO WHILE konum > 0
        adet += 1
        konum = INSTR(konum + LEN(parca), metin, parca)
    LOOP
    RETURN adet
END FUNCTION
PRINT Say("[INFO] a [WARN] b [INFO] c", "[INFO]")
