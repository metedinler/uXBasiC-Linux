K6_sinif_yontemi_yalin_cagri
' Sinif govdesinde yalin yontem cagrisi implicit THIS ile cozulur (semantic_implicit_this.fbs). Beklenen: 2
CLASS Sayac
    PUBLIC
    DIM n AS I32
    SUB Bump()
        n = n + 1
    END SUB
    SUB Iki()
        Bump()
        Bump()
    END SUB
END CLASS
DIM s AS Sayac = NEW Sayac()
s.Iki()
PRINT s.n
