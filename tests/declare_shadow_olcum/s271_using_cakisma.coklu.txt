U1_cakisma_ayni_ad_ayni_arite
' K3 karari (Mete, 2026-10-28): iki USING'ten gelen ayni ad, nitelensiz cagri = DERLEME HATASI.
' Hedef: altı yolda derleme hatasi, "once" HIC yazilmaz. Bugun: sessiz ilk-kazanan (7).
NAMESPACE A
    FUNCTION Topla(x AS I32, y AS I32) AS I32
        RETURN x + y
    END FUNCTION
END NAMESPACE
NAMESPACE B
    FUNCTION Topla(x AS I32, y AS I32) AS I32
        RETURN x * y
    END FUNCTION
END NAMESPACE
USING A
USING B
PRINT "once"
PRINT Topla(3, 4)

U8_namespace_ici_kendi_adi
' NAMESPACE A'nin gövdesinde nitelensiz Topla kendi namespace'ine gider (S-141), B'nin USING'i belirsizlik uretmez.
' Beklenen: 7
NAMESPACE B
    FUNCTION Topla(x AS I32, y AS I32) AS I32
        RETURN x * y
    END FUNCTION
END NAMESPACE
USING B
NAMESPACE A
    FUNCTION Topla(x AS I32, y AS I32) AS I32
        RETURN x + y
    END FUNCTION
    FUNCTION Cagir(x AS I32, y AS I32) AS I32
        RETURN Topla(x, y)
    END FUNCTION
END NAMESPACE
PRINT A.Cagir(3, 4)
