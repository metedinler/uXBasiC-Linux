W1_sarmalayici_ic_cagri_yalin
' Mete: "Left sonucuna [ ] ekleyen" yerlesik yetenek ornegi. NAMESPACE icinde yalin Left(...) cagrisi
' kendini mi (sonsuz ozyineleme), yerlesigi mi cagirir? Beklenen: [Mer] / Mer
NAMESPACE S
    FUNCTION Left(s AS STRING, n AS I32) AS STRING
        RETURN "[" + Left(s, n) + "]"
    END FUNCTION
END NAMESPACE
PRINT S.Left("Merhaba", 3)
PRINT Left("Merhaba", 3)

W4_sarmalayici_global_ad_kok
