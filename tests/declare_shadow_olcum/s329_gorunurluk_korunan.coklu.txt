P1_ad_alani_icinden_kok_yordam
' Ad alani icinde nitelensiz cagri, ust duzey (kok) yordama gider. Beklenen: 4
FUNCTION Kok(a AS I32) AS I32
    RETURN a + 1
END FUNCTION
NAMESPACE A
    FUNCTION Say(a AS I32) AS I32
        RETURN Kok(a)
    END FUNCTION
END NAMESPACE
PRINT A.Say(3)

P8_sinif_alan_dizisi
' Sinif govdesinde dizi alani a(i) okunur. Beklenen: 5
CLASS Kutu
    PUBLIC
    DIM d(3) AS I32
    SUB Koy()
        d(1) = 5
    END SUB
    SUB Yaz()
        PRINT d(1)
    END SUB
END CLASS
DIM k AS Kutu = NEW Kutu()
k.Koy()
k.Yaz()
=====

P10_yontem_nesne_uzerinden
' Yontem yalnizca nesne uzerinden nitelikli cagrilir. Beklenen: 1
CLASS Sayac
    PUBLIC
    DIM n AS I32
    SUB Bump()
        n = n + 1
    END SUB
END CLASS
DIM s AS Sayac = NEW Sayac()
s.Bump()
PRINT s.n
